function genFizzBuzz(rules, num) {

JV genFizzBuzz 
public static String genFizzBuzz(List<Rule> rules, int num) {
    StringBuilder sb = new StringBuilder();
    for (Rule r : rules) {
        if (num % r.factor == 0) sb.append(r.word);
    }
    return sb.length() > 0 ? sb.toString() : String.valueOf(num);
}
